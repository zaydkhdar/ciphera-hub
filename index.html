async function searchGitHub() { const query = document.getElementById('searchInput').value || "python"; const grid = document.getElementById('toolGrid'); const loader = document.getElementById('loader'); grid.innerHTML = ''; loader.style.display = 'block'; try { // THE CACHE BUSTER: &t=${Date.now()} forces a fresh search every time const res = await fetch(`/api/search?q=${encodeURIComponent(query)}&t=${Date.now()}`); const data = await res.json(); loader.style.display = 'none'; if (!res.ok) throw new Error(data.error || "Failed to load"); if (data.length === 0) throw new Error("No repositories found."); data.forEach(repo => { const card = document.createElement('div'); card.className = 'tool-card'; card.innerHTML = `
${repo.name}
@${repo.owner}
${repo.description}
🔥 Heat Index: ${repo.ai_score}/100
${repo.ai_reason}
⭐ ${repo.stars.toLocaleString()}
View Repo ↗
`; grid.appendChild(card); }); } catch (err) { loader.style.display = 'none'; grid.innerHTML = `

SYSTEM ERROR

${err.message}

`; } }
